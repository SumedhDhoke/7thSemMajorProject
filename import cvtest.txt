import cv2
import numpy as np
import onnxruntime as ort
from picamera2 import Picamera2
import sys
from collections import deque, Counter


# ==========================================
# FILE PATHS
# ==========================================

EMOTION_MODEL = "/home/majorproject/Downloads/emotion_model.onnx"

SSD_PROTOTXT = (
    "/home/majorproject/Downloads/mobilenet/"
    "MobileNetSSD_deploy.prototxt"
)

SSD_MODEL = (
    "/home/majorproject/Downloads/mobilenet/"
    "MobileNetSSD_deploy.caffemodel"
)

CASCADE = (
    "/usr/share/opencv4/haarcascades/"
    "haarcascade_frontalface_default.xml"
)


# ==========================================
# EMOTION MODEL
# ==========================================

try:

    emotion_net = ort.InferenceSession(
        EMOTION_MODEL,
        providers=["CPUExecutionProvider"]
    )

    emotion_input = emotion_net.get_inputs()[0].name
    emotion_output = emotion_net.get_outputs()[0].name

    print("Emotion model: OK")

except Exception as e:

    print("Emotion model error:")
    print(e)
    sys.exit()


# ==========================================
# FER+ EMOTION LABELS
# ==========================================

emotions = [
    "neutral",
    "happy",
    "surprise",
    "sad",
    "anger",
    "disgust",
    "fear",
    "contempt"
]


# ==========================================
# MAP EMOTIONS FOR ROBOT
# ==========================================

def get_robot_emotion(result):

    scores = result[0]

    emotion_id = np.argmax(scores)

    emotion = emotions[emotion_id]

    # Simplify emotions for robot
    if emotion == "disgust":
        emotion = "anger"

    elif emotion == "fear":
        emotion = "surprise"

    elif emotion == "contempt":
        emotion = "neutral"

    return emotion


# ==========================================
# EMOTION SMOOTHING
# ==========================================

emotion_history = deque(maxlen=7)


def smooth_emotion(emotion):

    emotion_history.append(emotion)

    counts = Counter(
        emotion_history
    )

    return counts.most_common(1)[0][0]


# ==========================================
# FACE DETECTOR
# ==========================================

face_detector = cv2.CascadeClassifier(
    CASCADE
)

if face_detector.empty():

    print("Face detector error!")
    print("Check path:")
    print(CASCADE)

    sys.exit()

print("Face detector: OK")


# ==========================================
# MOBILE NET SSD
# ==========================================

try:

    object_net = cv2.dnn.readNetFromCaffe(
        SSD_PROTOTXT,
        SSD_MODEL
    )

    print("MobileNet-SSD: OK")

except Exception as e:

    print("MobileNet-SSD error:")
    print(e)
    sys.exit()


# ==========================================
# MOBILE NET SSD CLASSES
# ==========================================

classes = [
    "background",
    "aeroplane",
    "bicycle",
    "bird",
    "boat",
    "bottle",
    "bus",
    "car",
    "cat",
    "chair",
    "cow",
    "diningtable",
    "dog",
    "horse",
    "motorbike",
    "person",
    "pottedplant",
    "sheep",
    "sofa",
    "train",
    "tvmonitor"
]


# ==========================================
# CAMERA
# ==========================================

try:

    camera = Picamera2()

    config = camera.create_preview_configuration(
        main={
            "size": (640, 480),
            "format": "RGB888"
        }
    )

    camera.configure(config)
    camera.start()

    print("Camera: OK")

except Exception as e:

    print("Camera error:")
    print(e)
    sys.exit()


print()
print("==============================")
print(" AI COMPANION STARTED")
print("==============================")
print("Press Q to quit.")
print()


# ==========================================
# VARIABLES
# ==========================================

frame_count = 0

# Object detection every 5 frames
OBJECT_INTERVAL = 5

last_objects = []

current_emotion = "neutral"


# ==========================================
# MAIN LOOP
# ==========================================

while True:

    # ======================================
    # CAPTURE FRAME
    # ======================================

    frame = camera.capture_array()

    # RGB → BGR
    frame = cv2.cvtColor(
        frame,
        cv2.COLOR_RGB2BGR
    )

    frame_count += 1


    # ======================================
    # FACE DETECTION
    # ======================================

    gray = cv2.cvtColor(
        frame,
        cv2.COLOR_BGR2GRAY
    )

    faces = face_detector.detectMultiScale(
        gray,
        scaleFactor=1.2,
        minNeighbors=5,
        minSize=(60, 60)
    )


    # ======================================
    # EMOTION DETECTION
    # ======================================

    for (x, y, w, h) in faces:

        # Extract face
        face = gray[
            y:y+h,
            x:x+w
        ]

        # Resize for FER+
        face = cv2.resize(
            face,
            (64, 64)
        )

        # Normalize
        face = face.astype(
            np.float32
        ) / 255.0

        # Shape:
        # 1 × 1 × 64 × 64

        face = np.expand_dims(
            face,
            axis=0
        )

        face = np.expand_dims(
            face,
            axis=0
        )

        # Run emotion model
        result = emotion_net.run(
            [emotion_output],
            {
                emotion_input: face
            }
        )[0]

        # Get robot emotion
        detected_emotion = get_robot_emotion(
            result
        )

        # Smooth emotion
        current_emotion = smooth_emotion(
            detected_emotion
        )

        # Print emotion
        print(
            "Emotion:",
            current_emotion
        )

        # ==================================
        # FACE BOX
        # ==================================

        cv2.rectangle(
            frame,
            (x, y),
            (x+w, y+h),
            (0, 255, 0),
            2
        )

        # ==================================
        # EMOTION LABEL
        # ==================================

        cv2.putText(
            frame,
            "Emotion: " + current_emotion,
            (x, y - 10),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            (0, 255, 0),
            2
        )


    # ======================================
    # OBJECT DETECTION
    # ======================================

    if frame_count % OBJECT_INTERVAL == 0:

        # Create MobileNet-SSD input
        blob = cv2.dnn.blobFromImage(
            frame,
            scalefactor=0.007843,
            size=(300, 300),
            mean=127.5,
            swapRB=False,
            crop=False
        )

        object_net.setInput(
            blob
        )

        detections = object_net.forward()

        last_objects = []

        height, width = frame.shape[:2]


        # ==================================
        # PROCESS DETECTIONS
        # ==================================

        for i in range(
            detections.shape[2]
        ):

            confidence = detections[
                0, 0, i, 2
            ]

            # Confidence threshold
            if confidence < 0.50:
                continue

            class_id = int(
                detections[
                    0, 0, i, 1
                ]
            )

            if (
                class_id < 0
                or
                class_id >= len(classes)
            ):
                continue

            object_name = classes[
                class_id
            ]

            # Save object
            last_objects.append(
                object_name
            )

            # ==================================
            # BOUNDING BOX
            # ==================================

            box = detections[
                0, 0, i, 3:7
            ]

            box = box * np.array(
                [
                    width,
                    height,
                    width,
                    height
                ]
            )

            x1, y1, x2, y2 = box.astype(
                int
            )

            # Keep inside frame
            x1 = max(
                0,
                x1
            )

            y1 = max(
                0,
                y1
            )

            x2 = min(
                width,
                x2
            )

            y2 = min(
                height,
                y2
            )

            # ==================================
            # DRAW OBJECT BOX
            # ==================================

            cv2.rectangle(
                frame,
                (x1, y1),
                (x2, y2),
                (255, 0, 0),
                2
            )

            # ==================================
            # OBJECT LABEL
            # ==================================

            label = (
                object_name
                + " "
                + str(
                    round(
                        float(confidence),
                        2
                    )
                )
            )

            cv2.putText(
                frame,
                label,
                (
                    x1,
                    max(
                        y1 - 5,
                        20
                    )
                ),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.6,
                (255, 0, 0),
                2
            )


    # ======================================
    # TERMINAL OBJECT OUTPUT
    # ======================================

    if last_objects:

        unique_objects = list(
            set(last_objects)
        )

        print(
            "Objects:",
            ", ".join(
                unique_objects
            )
        )


    # ======================================
    # DISPLAY CURRENT EMOTION
    # ======================================

    cv2.putText(
        frame,
        "Robot Emotion: " + current_emotion,
        (10, 30),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (0, 255, 0),
        2
    )


    # ======================================
    # DISPLAY CAMERA
    # ======================================

    cv2.imshow(
        "Raspberry Pi AI Companion",
        frame
    )


    # ======================================
    # QUIT
    # ======================================

    if cv2.waitKey(1) & 0xFF == ord("q"):

        break


# ==========================================
# CLEANUP
# ==========================================

camera.stop()

cv2.destroyAllWindows()

print()
print("Program stopped.")
