import cv2
import numpy as np
import onnxruntime as ort
from picamera2 import Picamera2
import sys
from collections import deque, Counter


# ============================================================
# FILE PATHS
# ============================================================

EMOTION_MODEL = "/home/majorproject/Downloads/emotion_model.onnx"

SSD_PROTOTXT = (
    "/home/majorproject/Downloads/mobilenet/"
    "MobileNetSSD_deploy.prototxt"
)

SSD_MODEL = (
    "/home/majorproject/Downloads/mobilenet/"
    "MobileNetSSD_deploy.caffemodel"
)

CASCADE = (
    "/usr/share/opencv4/haarcascades/"
    "haarcascade_frontalface_default.xml"
)


# ============================================================
# EMOTION MODEL
# ============================================================

try:

    emotion_net = ort.InferenceSession(
        EMOTION_MODEL,
        providers=["CPUExecutionProvider"]
    )

    emotion_input = emotion_net.get_inputs()[0].name
    emotion_output = emotion_net.get_outputs()[0].name

    print("Emotion model loaded successfully.")

except Exception as e:

    print("ERROR: Emotion model could not be loaded.")
    print(e)
    sys.exit()


# ============================================================
# FER+ EMOTIONS
# ============================================================

emotions = [
    "Neutral",
    "Happy",
    "Surprise",
    "Sad",
    "Angry",
    "Disgust",
    "Fear",
    "Contempt"
]


# ============================================================
# SOFTMAX
# ============================================================

def softmax(x):

    x = x - np.max(x)

    exp = np.exp(x)

    return exp / np.sum(exp)


# ============================================================
# EMOTION SMOOTHING
# ============================================================

emotion_history = deque(maxlen=5)


def smooth_emotion(emotion):

    emotion_history.append(emotion)

    count = Counter(
        emotion_history
    )

    return count.most_common(1)[0][0]


# ============================================================
# FACE DETECTOR
# ============================================================

face_detector = cv2.CascadeClassifier(
    CASCADE
)

if face_detector.empty():

    print("ERROR: Face detector could not be loaded.")
    print("Path:", CASCADE)

    sys.exit()

print("Face detector loaded successfully.")


# ============================================================
# MOBILE-NET SSD
# ============================================================

try:

    object_net = cv2.dnn.readNetFromCaffe(
        SSD_PROTOTXT,
        SSD_MODEL
    )

    print("MobileNet-SSD loaded successfully.")

except Exception as e:

    print("ERROR: MobileNet-SSD could not be loaded.")
    print(e)

    sys.exit()


# ============================================================
# OBJECT CLASSES
# ============================================================

classes = [
    "background",
    "aeroplane",
    "bicycle",
    "bird",
    "boat",
    "bottle",
    "bus",
    "car",
    "cat",
    "chair",
    "cow",
    "diningtable",
    "dog",
    "horse",
    "motorbike",
    "person",
    "pottedplant",
    "sheep",
    "sofa",
    "train",
    "tvmonitor"
]


# ============================================================
# CAMERA
# ============================================================

try:

    camera = Picamera2()

    config = camera.create_preview_configuration(
        main={
            "size": (640, 480),
            "format": "RGB888"
        }
    )

    camera.configure(config)

    camera.start()

    print("Camera started successfully.")

except Exception as e:

    print("ERROR: Camera could not start.")
    print(e)

    sys.exit()


print()
print("--------------------------------")
print(" AI COMPANION STARTED")
print("--------------------------------")
print("Press Q to quit.")
print()


# ============================================================
# VARIABLES
# ============================================================

frame_count = 0

OBJECT_INTERVAL = 8

current_emotion = "Neutral"

last_objects = []


# ============================================================
# MAIN LOOP
# ============================================================

while True:

    # --------------------------------------------------------
    # CAPTURE ORIGINAL RGB FRAME
    # --------------------------------------------------------

    rgb_frame = camera.capture_array()

    frame_count += 1


    # --------------------------------------------------------
    # CREATE BGR COPY ONLY FOR OPENCV DISPLAY
    # --------------------------------------------------------

    frame = cv2.cvtColor(
        rgb_frame,
        cv2.COLOR_RGB2BGR
    )


    # ========================================================
    # FACE DETECTION
    # ========================================================

    gray = cv2.cvtColor(
        frame,
        cv2.COLOR_BGR2GRAY
    )


    faces = face_detector.detectMultiScale(
        gray,
        scaleFactor=1.15,
        minNeighbors=6,
        minSize=(70, 70)
    )


    # ========================================================
    # EMOTION DETECTION
    # ========================================================

    for (x, y, w, h) in faces:

        # ----------------------------------------------------
        # Extract face
        # ----------------------------------------------------

        face = gray[
            y:y+h,
            x:x+w
        ]


        if face.size == 0:

            continue


        # ----------------------------------------------------
        # Resize to FER+ input
        # ----------------------------------------------------

        face = cv2.resize(
            face,
            (64, 64),
            interpolation=cv2.INTER_AREA
        )


        # ----------------------------------------------------
        # IMPORTANT:
        #
        # FER+ expects grayscale pixel values.
        # Do NOT divide by 255.
        # ----------------------------------------------------

        face = face.astype(
            np.float32
        )


        # ----------------------------------------------------
        # Shape:
        #
        # 64 x 64
        #       ↓
        # 1 x 64 x 64
        #       ↓
        # 1 x 1 x 64 x 64
        # ----------------------------------------------------

        face = np.expand_dims(
            face,
            axis=0
        )

        face = np.expand_dims(
            face,
            axis=0
        )


        # ----------------------------------------------------
        # Run FER+ model
        # ----------------------------------------------------

        result = emotion_net.run(
            [emotion_output],
            {
                emotion_input: face
            }
        )[0]


        # ----------------------------------------------------
        # Convert scores to probabilities
        # ----------------------------------------------------

        probabilities = softmax(
            result[0]
        )


        # ----------------------------------------------------
        # Find strongest emotion
        # ----------------------------------------------------

        emotion_id = np.argmax(
            probabilities
        )

        detected_emotion = emotions[
            emotion_id
        ]


        # ----------------------------------------------------
        # Smooth emotion
        # ----------------------------------------------------

        current_emotion = smooth_emotion(
            detected_emotion
        )


        confidence = probabilities[
            emotion_id
        ] * 100


        # ----------------------------------------------------
        # Terminal output
        # ----------------------------------------------------

        print(
            "Emotion:",
            current_emotion,
            "| Confidence:",
            round(
                float(confidence),
                1
            ),
            "%"
        )


        # ====================================================
        # DRAW FACE BOX
        # ====================================================

        cv2.rectangle(
            frame,
            (x, y),
            (x+w, y+h),
            (0, 255, 0),
            2
        )


        # ====================================================
        # EMOTION TEXT
        # ====================================================

        text = (
            current_emotion
            + " "
            + str(
                round(
                    float(confidence),
                    1
                )
            )
            + "%"
        )


        cv2.putText(
            frame,
            text,
            (x, y - 10),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.65,
            (0, 255, 0),
            2
        )


    # ========================================================
    # OBJECT DETECTION
    # ========================================================

    if frame_count % OBJECT_INTERVAL == 0:

        blob = cv2.dnn.blobFromImage(
            frame,
            scalefactor=0.007843,
            size=(300, 300),
            mean=127.5,
            swapRB=False,
            crop=False
        )


        object_net.setInput(
            blob
        )


        detections = object_net.forward()


        last_objects = []


        height, width = frame.shape[:2]


        # ====================================================
        # PROCESS OBJECTS
        # ====================================================

        for i in range(
            detections.shape[2]
        ):

            confidence = detections[
                0, 0, i, 2
            ]


            if confidence < 0.50:

                continue


            class_id = int(
                detections[
                    0, 0, i, 1
                ]
            )


            if (
                class_id < 0
                or
                class_id >= len(classes)
            ):

                continue


            object_name = classes[
                class_id
            ]


            last_objects.append(
                object_name
            )


            # ------------------------------------------------
            # Bounding box
            # ------------------------------------------------

            box = detections[
                0, 0, i, 3:7
            ]


            box = box * np.array(
                [
                    width,
                    height,
                    width,
                    height
                ]
            )


            x1, y1, x2, y2 = (
                box.astype(int)
            )


            x1 = max(
                0,
                x1
            )

            y1 = max(
                0,
                y1
            )

            x2 = min(
                width,
                x2
            )

            y2 = min(
                height,
                y2
            )


            # ------------------------------------------------
            # Draw object box
            # ------------------------------------------------

            cv2.rectangle(
                frame,
                (x1, y1),
                (x2, y2),
                (255, 0, 0),
                2
            )


            # ------------------------------------------------
            # Object label
            # ------------------------------------------------

            label = (
                object_name
                + " "
                + str(
                    round(
                        float(confidence),
                        2
                    )
                )
            )


            cv2.putText(
                frame,
                label,
                (
                    x1,
                    max(
                        y1 - 5,
                        20
                    )
                ),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.55,
                (255, 0, 0),
                2
            )


    # ========================================================
    # TOP STATUS BAR
    # ========================================================

    cv2.putText(
        frame,
        "Emotion: " + current_emotion,
        (10, 30),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7,
        (0, 255, 0),
        2
    )


    # ========================================================
    # DISPLAY
    # ========================================================

    cv2.imshow(
        "AI Companion",
        frame
    )


    # ========================================================
    # QUIT
    # ========================================================

    if cv2.waitKey(1) & 0xFF == ord("q"):

        break


# ============================================================
# CLEANUP
# ============================================================

camera.stop()

cv2.destroyAllWindows()

print()
print("Program stopped.")
