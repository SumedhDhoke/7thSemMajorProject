import cv2
import numpy as np
import onnxruntime as ort
from picamera2 import Picamera2
import sys
import time


# ==========================================
# 1. EMOTION MODEL
# ==========================================

EMOTION_MODEL = "/home/majorproject/Downloads/emotion_model.onnx"

try:
    emotion_session = ort.InferenceSession(
        EMOTION_MODEL,
        providers=["CPUExecutionProvider"]
    )

    emotion_input = emotion_session.get_inputs()[0].name
    emotion_output = emotion_session.get_outputs()[0].name

    print("Emotion model loaded successfully.")

except Exception as e:
    print("ERROR: Could not load emotion model.")
    print(e)
    sys.exit()


emotions = [
    "neutral",
    "happy",
    "surprise",
    "sad",
    "anger",
    "disgust",
    "fear",
    "contempt"
]


# ==========================================
# 2. MOBILENET-SSD OBJECT DETECTOR
# ==========================================

SSD_CONFIG = (
    "/home/majorproject/Downloads/mobilenet/"
    "MobileNetSSD_deploy.prototxt"
)

SSD_MODEL = (
    "/home/majorproject/Downloads/mobilenet/"
    "MobileNetSSD_deploy.caffemodel"
)

try:

    object_net = cv2.dnn.readNetFromCaffe(
        SSD_CONFIG,
        SSD_MODEL
    )

    print("MobileNet-SSD loaded successfully.")

except Exception as e:

    print("ERROR: Could not load MobileNet-SSD.")
    print(e)
    sys.exit()


# MobileNet-SSD classes
classes = [
    "background",
    "aeroplane",
    "bicycle",
    "bird",
    "boat",
    "bottle",
    "bus",
    "car",
    "cat",
    "chair",
    "cow",
    "diningtable",
    "dog",
    "horse",
    "motorbike",
    "person",
    "pottedplant",
    "sheep",
    "sofa",
    "train",
    "tvmonitor"
]


# ==========================================
# 3. FACE DETECTOR
# ==========================================

CASCADE_PATH = (
    "/usr/share/opencv4/haarcascades/"
    "haarcascade_frontalface_default.xml"
)

face_cascade = cv2.CascadeClassifier(
    CASCADE_PATH
)

if face_cascade.empty():

    print("ERROR: Face detector could not be loaded!")
    print("Path:", CASCADE_PATH)

    sys.exit()

print("Face detector loaded successfully.")


# ==========================================
# 4. CAMERA
# ==========================================

picam2 = Picamera2()

camera_config = picam2.create_preview_configuration(
    main={
        "size": (640, 480),
        "format": "RGB888"
    }
)

picam2.configure(camera_config)
picam2.start()

print("Camera started.")
print("Press Q to quit.")


# ==========================================
# 5. VARIABLES
# ==========================================

frame_count = 0

# Run object detection every 5 frames
OBJECT_INTERVAL = 5

last_objects = []


# ==========================================
# 6. MAIN LOOP
# ==========================================

while True:

    frame = picam2.capture_array()

    frame_count += 1

    # ======================================
    # FACE + EMOTION DETECTION
    # ======================================

    gray = cv2.cvtColor(
        frame,
        cv2.COLOR_RGB2GRAY
    )

    faces = face_cascade.detectMultiScale(
        gray,
        scaleFactor=1.2,
        minNeighbors=5,
        minSize=(60, 60)
    )

    for (x, y, w, h) in faces:

        face = gray[
            y:y+h,
            x:x+w
        ]

        face = cv2.resize(
            face,
            (64, 64)
        )

        face = face.astype(
            np.float32
        ) / 255.0

        face = np.expand_dims(
            face,
            axis=0
        )

        face = np.expand_dims(
            face,
            axis=0
        )

        result = emotion_session.run(
            [emotion_output],
            {emotion_input: face}
        )[0]

        emotion_index = np.argmax(result)

        emotion = emotions[
            emotion_index
        ]

        # Draw face
        cv2.rectangle(
            frame,
            (x, y),
            (x+w, y+h),
            (0, 255, 0),
            2
        )

        cv2.putText(
            frame,
            "Emotion: " + emotion,
            (x, y - 10),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.6,
            (0, 255, 0),
            2
        )


    # ======================================
    # OBJECT DETECTION
    # ======================================

    # Only run every 5 frames
    if frame_count % OBJECT_INTERVAL == 0:

        blob = cv2.dnn.blobFromImage(
            frame,
            scalefactor=0.007843,
            size=(300, 300),
            mean=127.5,
            swapRB=False,
            crop=False
        )

        object_net.setInput(blob)

        detections = object_net.forward()

        last_objects = []

        for i in range(
            detections.shape[2]
        ):

            confidence = detections[
                0, 0, i, 2
            ]

            if confidence < 0.50:
                continue

            class_id = int(
                detections[
                    0, 0, i, 1
                ]
            )

            if class_id >= len(classes):
                continue

            object_name = classes[
                class_id
            ]

            last_objects.append(
                object_name
            )

            # Bounding box
            h, w = frame.shape[:2]

            box = detections[
                0, 0, i, 3:7
            ] * np.array(
                [w, h, w, h]
            )

            x1, y1, x2, y2 = box.astype(
                int
            )

            x1 = max(0, x1)
            y1 = max(0, y1)
            x2 = min(w, x2)
            y2 = min(h, y2)

            # Draw object
            cv2.rectangle(
                frame,
                (x1, y1),
                (x2, y2),
                (255, 0, 0),
                2
            )

            label = (
                object_name
                + " "
                + str(round(float(confidence), 2))
            )

            cv2.putText(
                frame,
                label,
                (x1, y1 - 5),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.55,
                (255, 0, 0),
                2
            )


    # ======================================
    # TERMINAL OUTPUT
    # ======================================

    if last_objects:

        print(
            "Objects:",
            ", ".join(
                set(last_objects)
            )
        )


    # ======================================
    # DISPLAY
    # ======================================

    cv2.imshow(
        "AI Companion",
        frame
    )

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break


# ==========================================
# 7. CLEANUP
# ==========================================

picam2.stop()

cv2.destroyAllWindows()

print("Program stopped.")
