import cv2
import numpy as np
import onnxruntime as ort
from picamera2 import Picamera2
import sys


# ==========================================
# FILE PATHS
# ==========================================

EMOTION_MODEL = "/home/majorproject/Downloads/emotion_model.onnx"

SSD_PROTOTXT = "/home/majorproject/Downloads/mobilenet/MobileNetSSD_deploy.prototxt"

SSD_MODEL = "/home/majorproject/Downloads/mobilenet/MobileNetSSD_deploy.caffemodel"

CASCADE = "/usr/share/opencv4/haarcascades/haarcascade_frontalface_default.xml"


# ==========================================
# EMOTION MODEL
# ==========================================

try:

    emotion_net = ort.InferenceSession(
        EMOTION_MODEL,
        providers=["CPUExecutionProvider"]
    )

    emotion_input = emotion_net.get_inputs()[0].name
    emotion_output = emotion_net.get_outputs()[0].name

    print("Emotion model: OK")

except Exception as e:

    print("Emotion model error:")
    print(e)
    sys.exit()


# ==========================================
# EMOTIONS
# ==========================================

emotions = [
    "neutral",
    "happy",
    "surprise",
    "sad",
    "anger",
    "disgust",
    "fear",
    "contempt"
]


# ==========================================
# FACE DETECTOR
# ==========================================

face_detector = cv2.CascadeClassifier(CASCADE)

if face_detector.empty():

    print("Face detector error!")
    print("Check:", CASCADE)

    sys.exit()

print("Face detector: OK")


# ==========================================
# MOBILE NET SSD
# ==========================================

try:

    object_net = cv2.dnn.readNetFromCaffe(
        SSD_PROTOTXT,
        SSD_MODEL
    )

    print("MobileNet-SSD: OK")

except Exception as e:

    print("MobileNet-SSD error:")
    print(e)
    sys.exit()


# ==========================================
# OBJECT CLASSES
# ==========================================

classes = [
    "background",
    "aeroplane",
    "bicycle",
    "bird",
    "boat",
    "bottle",
    "bus",
    "car",
    "cat",
    "chair",
    "cow",
    "diningtable",
    "dog",
    "horse",
    "motorbike",
    "person",
    "pottedplant",
    "sheep",
    "sofa",
    "train",
    "tvmonitor"
]


# ==========================================
# CAMERA
# ==========================================

try:

    camera = Picamera2()

    config = camera.create_preview_configuration(
        main={
            "size": (640, 480),
            "format": "RGB888"
        }
    )

    camera.configure(config)
    camera.start()

    print("Camera: OK")

except Exception as e:

    print("Camera error:")
    print(e)
    sys.exit()


print("\nSystem started.")
print("Press Q to quit.")


# ==========================================
# MAIN LOOP
# ==========================================

while True:

    # Capture image
    frame = camera.capture_array()

    # Convert RGB → BGR
    frame = cv2.cvtColor(
        frame,
        cv2.COLOR_RGB2BGR
    )


    # ======================================
    # FACE DETECTION
    # ======================================

    gray = cv2.cvtColor(
        frame,
        cv2.COLOR_BGR2GRAY
    )

    faces = face_detector.detectMultiScale(
        gray,
        scaleFactor=1.2,
        minNeighbors=5,
        minSize=(60, 60)
    )


    # ======================================
    # EMOTION DETECTION
    # ======================================

    for (x, y, w, h) in faces:

        face = gray[
            y:y+h,
            x:x+w
        ]

        face = cv2.resize(
            face,
            (64, 64)
        )

        face = face.astype(
            np.float32
        ) / 255.0

        face = np.expand_dims(
            face,
            axis=0
        )

        face = np.expand_dims(
            face,
            axis=0
        )

        result = emotion_net.run(
            [emotion_output],
            {emotion_input: face}
        )[0]

        emotion_id = np.argmax(result)

        emotion = emotions[
            emotion_id
        ]

        print(
            "Emotion:",
            emotion
        )

        # Face box
        cv2.rectangle(
            frame,
            (x, y),
            (x+w, y+h),
            (0, 255, 0),
            2
        )

        # Emotion text
        cv2.putText(
            frame,
            emotion,
            (x, y-10),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            (0, 255, 0),
            2
        )


    # ======================================
    # OBJECT DETECTION
    # ======================================

    blob = cv2.dnn.blobFromImage(
        frame,
        scalefactor=0.007843,
        size=(300, 300),
        mean=127.5,
        swapRB=False,
        crop=False
    )

    object_net.setInput(blob)

    detections = object_net.forward()


    # ======================================
    # PROCESS OBJECTS
    # ======================================

    height, width = frame.shape[:2]

    for i in range(
        detections.shape[2]
    ):

        confidence = detections[
            0, 0, i, 2
        ]

        # Ignore weak detections
        if confidence < 0.50:
            continue

        class_id = int(
            detections[
                0, 0, i, 1
            ]
        )

        if class_id < 0 or class_id >= len(classes):
            continue

        object_name = classes[
            class_id
        ]

        # Bounding box
        box = detections[
            0, 0, i, 3:7
        ]

        box = box * np.array(
            [width, height, width, height]
        )

        x1, y1, x2, y2 = box.astype(int)

        # Keep inside image
        x1 = max(0, x1)
        y1 = max(0, y1)
        x2 = min(width, x2)
        y2 = min(height, y2)

        print(
            "Object:",
            object_name,
            "Confidence:",
            round(float(confidence), 2)
        )

        # Object box
        cv2.rectangle(
            frame,
            (x1, y1),
            (x2, y2),
            (255, 0, 0),
            2
        )

        # Object name
        cv2.putText(
            frame,
            object_name,
            (x1, max(y1-5, 20)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.6,
            (255, 0, 0),
            2
        )


    # ======================================
    # DISPLAY
    # ======================================

    cv2.imshow(
        "Raspberry Pi AI Companion",
        frame
    )

    # Q = quit
    if cv2.waitKey(1) & 0xFF == ord("q"):
        break


# ==========================================
# CLEANUP
# ==========================================

camera.stop()

cv2.destroyAllWindows()

print("Program stopped.")
